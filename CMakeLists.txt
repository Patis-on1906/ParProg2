cmake_minimum_required(VERSION 3.16)
project(DelegationFactorization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(delegation main.cpp)
target_link_libraries(delegation PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(delegation PRIVATE /W4)
else()
    target_compile_options(delegation PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
